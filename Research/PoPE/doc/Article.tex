\section{Introduction}
Standard quantum mechanics treats superposition as ontic and collapse as a probabilistic projection often tied to observation. Relativity treats space and time as geometric backgrounds shaped by energy. These frameworks remain disconnected and struggle with dark matter, dark energy, and the origin of time.

We unify these domains by reinterpreting collapse as the primary physical process. Energy always collapses; persistence emerges only via \emph{anchoring}. Space, time, and matter arise statistically from successful persistence. This \emph{collapse--churn} picture reframes quantum indeterminacy and relativistic geometry as consequences of the same substrate dynamics.

% <!-- [SS-5]: Abstract ----->
\vspace{5em}
\begin{abstract}
    \noindent We propose a set of postulates unifying quantum measurement, relativity, and cosmology under a single dynamical principle: \emph{\textcolor{primaryblue}{continuous collapse and recoherence of energy}}.
    Collapse is not observer-induced but universal and timeless.
    Persistence arises only when collapses mutually anchor, yielding emergent space, time, and stable matter. Observation is redefined as an emergent form of anchoring, possible only after initial collapses have persisted.
    Time dilation, gravity, and dark matter follow naturally as statistical consequences of collapse--recoherence churn.
    The framework is specified via CSL-type dynamics with causal rate modulation by an anchor field, a geometry-first postulate, retarded Green's functions, and persistence thresholds, leading to falsifiable astrophysical and laboratory predictions.
\end{abstract}

\section{Postulates}

% /6.1/ Postulate 1: Universal Collapse
\begin{postulate}{Postulate 1: Universal Collapse}
    All energy collapses continuously and universally into configurations permitted by its probability structure. State evolution follows a CSL-type It\^o SDE:
    \begin{align}
    d|\psi_t\rangle&=\Big[-\tfrac{i}{\hbar}H\,dt-\tfrac12\!\int\!d^3x\,\Gamma(x)\,\big(\hat A(x)-\braket{\hat A(x)}_t\big)^2dt \nonumber\\
    &\qquad\qquad+\!\int\!d^3x\,\big(\hat A(x)-\braket{\hat A(x)}_t\big)\,dW_t(x)\Big]|\psi_t\rangle, \label{eq:SSE}
    \end{align}
    with white-noise correlations
    \begin{align}
    \mathbb E\big[dW_t(x)dW_t(y)\big]=\Gamma(x)\,\delta^{(3)}(x-y)\,dt,
    \end{align}
    and \emph{smeared} mass--density collapse operators
    \begin{align}
    \hat A(x)=\!\int\!d^3y\,g_\sigma(x-y)\,\hat M(y),\quad
    g_\sigma(r)=\frac{1}{(2\pi\sigma^2)^{3/2}}e^{-|r|^2/(2\sigma^2)}.
    \end{align}
    The local collapse rate is modulated by an anchor field $I(x)$:
    \begin{align}
    \Gamma(x)=\gamma_0\big[1+\lambda I(x)\big],\qquad \gamma_0>0,\ \lambda\ge 0. \label{eq:GammaDef}
    \end{align}
    If anchoring fails, collapses immediately recohere. Collapse frequencies follow the Born measure $|\alpha|^2$ (Gleason noncontextuality and quadratic weighting).
\end{postulate}

\section{Collapse dynamics (v1.1)}\label{sec:collapse}
Averaging \eqref{eq:SSE} yields the Lindblad master equation:
\begin{align}
\dot\rho=-\frac{i}{\hbar}[H,\rho]-\frac12\!\int\!d^3x\,\Gamma(x)\,[\hat A(x),[\hat A(x),\rho]]. \label{eq:Master}
\end{align}
This is completely positive and trace-preserving. The anchor coupling \eqref{eq:GammaDef} modifies rates only, preserving the quadratic structure and the Born rule.

\paragraph{Off-diagonal suppression.}
Let $\hat A_f=\int f(x)\hat A(x)\,d^3x$ with eigenbasis $\{\ket{a}\}$. Off-diagonals obey
\begin{align}
\frac{d}{dt}\rho_{ab}\sim -\frac12\!\int\!d^3x\,\Gamma(x)\,\big(A_a(x)-A_b(x)\big)^2\,\rho_{ab}\quad (a\neq b).
\end{align}
Pointer selection is toward (smeared) mass density eigenprojectors.

\paragraph{Norm/positivity and no-signaling.}
With local Hermitian Lindblad densities $L_x=\sqrt{\Gamma(x)}\hat A(x)$ and $[L_x,L_y]=0$ for spacelike separation, microcausality holds. If $\Gamma$ depends only on past-light-cone data via $I$ propagated by $G_{\mathrm{ret}}$, parameter independence is enforced and no controllable FTL arises.

\paragraph{Rates from anchoring.}

\begin{align}
\Gamma_{\mathrm{persist}}(x,t)&=\gamma_0\big[1+\lambda I(x,t)\big]\ \Hev\big(\Acal(x,t)-\Theta\big),\label{eq:GammaPersist}\\
\Gamma_{\mathrm{recohere}}(x,t)&=\gamma_0\big[1-\lambda I(x,t)\big]\ \Hev\big(\Theta-\Acal(x,t)\big),\label{eq:GammaRecoh}

\section{Conservation law (v1.3)}\label{sec:conservation}

\begin{align}
\frac{d}{dt}\braket{H}=\Tr(H\dot\rho)=-\frac12\!\int\!d^3x\,\Gamma(x)\,\Tr\!\big(H[\hat A,[\hat A,\rho]]\big)
=-\frac12\!\int\!d^3x\,\Gamma(x)\,\big\langle[\hat A,[H,\hat A]]\big\rangle.\label{eq:dHdt}
\end{align}
\emph{Churn-neutrality:} if $[\hat A,[H,\hat A]]=0$ (e.g.\ appropriate smearing and $H$ class), then $d\braket{H}/dt=0$ in isolation. Otherwise the rate is bounded $O(\gamma_0)$ (no GRW-style blow-up) for standard CSL parameters.

\begin{align}

\varepsilon_{\mathrm{tot}}=\varepsilon_{\mathrm{blue}}+\varepsilon_{\mathrm{shadow}},\ \mathbf j_{\mathrm{tot}}=\mathbf j_{\mathrm{blue}}+\mathbf j_{\mathrm{shadow}}.\label{eq:ContFinal}
